```ruby
# lib/onetime/models/webhook_delivery.rb
# frozen_string_literal: true
require 'uri'

module Onetime
  # Outbound user-webhook delivery log + per-customer roll-up (#4348).
  # Written best-effort by Operations::DispatchNotification#deliver_via_webhook;
  # read by GET /api/colonel/webhooks/deliveries, GetUserDetails and
  # `bin/ots webhooks deliveries`. Never raises into the dispatcher.
  class WebhookDelivery < Familia::Horreum
    prefix :webhook_delivery

    # Site-wide ring. member = JSON entry, score = `at` epoch (float).
    class_sorted_set :log

    MAX_LOG   = 500                     # scope: capped per-install ring
    STATS_TTL = 90 * 24 * 60 * 60       # per-customer roll-up self-cleans
    STATUSES  = %w[success skipped error].freeze
    SKIP_REASONS = %w[disabled].freeze
    MAX_HOST_LENGTH = 253

    class << self
      # @return [Hash, nil] the stored entry (string keys) or nil (never raises)
      def record(user_id:, url_host:, status:, reason: nil, error_class: nil,
                 http_status: nil, duration_ms: nil)
        status = status.to_s
        raise ArgumentError, "invalid status: #{status}" unless STATUSES.include?(status)

        entry = {
          'id'          => Familia.generate_id,
          'at'          => Familia.now,
          'user_id'     => user_id.nil? ? nil : user_id.to_s,       # PUBLIC extid, never objid
          'url_host'    => url_host.to_s[0, MAX_HOST_LENGTH],
          'status'      => status,
          'reason'      => reason&.to_s,
          'error_class' => error_class&.to_s,
          'http_status' => http_status&.to_i,
          'duration_ms' => duration_ms&.to_i,
        }
        log.add(entry, entry['at'])
        trim!
        bump_stats!(entry) unless entry['user_id'].nil?
        entry
      rescue StandardError => ex
        OT.le "[WebhookDelivery] record failed: #{ex.class}: #{ex.message}"
        nil
      end

      # Newest-first slice (EmailSuppression.recent_events idiom).
      def recent(limit = 50, offset = 0)
        limit = limit.to_i; offset = offset.to_i
        return [] if limit <= 0
        offset = 0 if offset.negative?
        log.revrange(offset, offset + limit - 1)
      end

      def count = log.element_count
      def trim!(cap = MAX_LOG) = log.remrangebyrank(0, -([cap.to_i, MAX_LOG].max + 1))

      # Host only — never path, query, userinfo or port. Bad URLs → 'invalid'.
      def url_host(url)
        host = URI.parse(url.to_s).host.to_s.downcase
        host.empty? ? 'invalid' : host[0, MAX_HOST_LENGTH]
      rescue URI::InvalidURIError
        'invalid'
      end

      def stats_key(extid) = "#{prefix}:#{extid}:stats"

      # @return [Hash, nil] typed roll-up or nil when the customer never delivered
      def stats_for(extid)
        raw = Familia.dbclient.hgetall(stats_key(extid))
        return nil if raw.nil? || raw.empty?
        {
          total: raw['total'].to_i, success: raw['success'].to_i,
          error: raw['error'].to_i, skipped: raw['skipped'].to_i,
          last_status: raw['last_status'], last_at: raw['last_at']&.to_f,
          last_host: raw['last_host'], last_error_class: raw['last_error_class'],
          last_http_status: raw['last_http_status'].to_s.empty? ? nil : raw['last_http_status'].to_i,
        }
      end

      def forget!(extid) = Familia.dbclient.del(stats_key(extid))

      private

      # Atomic counters + last-* stamps. Keep the dbclient resolve inline with
      # .multi (dispatch_notification.rb:128–131 explains why).
      def bump_stats!(entry)
        key = stats_key(entry['user_id'])
        Familia.dbclient.multi do |tx|
          tx.hincrby(key, 'total', 1)
          tx.hincrby(key, entry['status'], 1)
          tx.hset(key, 'last_status', entry['status'], 'last_at', entry['at'].to_s,
                       'last_host', entry['url_host'],
                       'last_error_class', entry['error_class'].to_s,
                       'last_http_status', entry['http_status'].to_s)
          tx.expire(key, STATS_TTL)
        end
      end
    end
  end
end
```

```ruby
        addressee   = @data[:addressee] || {}
        webhook_url = addressee[:webhook_url]

        unless webhook_url
          logger.debug 'No webhook_url for webhook notification, skipping'
          return :skipped                       # nothing attempted → nothing logged
        end

        customer = resolve_webhook_customer(addressee[:custid])
        user_id  = customer&.extid
        url_host = Onetime::WebhookDelivery.url_host(webhook_url)

        # Kill switch (#4348): checked BEFORE any DNS/HTTP work.
        if customer&.webhooks_disabled?
          logger.info 'Webhook delivery skipped: disabled for customer', custid: user_id
          Onetime::WebhookDelivery.record(user_id: user_id, url_host: url_host,
                                          status: 'skipped', reason: 'disabled')
          return :skipped
        end

        started = Process.clock_gettime(Process::CLOCK_MONOTONIC)
        payload  = build_webhook_payload
        response = send_webhook_request(webhook_url, payload)

        unless response.is_a?(Net::HTTPSuccess)
          raise WebhookResponseError.new(
            "Webhook returned #{response.code}: #{response.body&.slice(0, 200)}",
            http_status: response.code.to_i,
          )
        end

        Onetime::WebhookDelivery.record(user_id: user_id, url_host: url_host, status: 'success',
                                        http_status: response.code.to_i, duration_ms: elapsed_ms(started))
        logger.debug 'Webhook delivered', url: webhook_url, status: response.code
        :success
      rescue StandardError => ex
        Onetime::WebhookDelivery.record(
          user_id: user_id, url_host: url_host, status: 'error',
          error_class: ex.class.name,
          http_status: ex.respond_to?(:http_status) ? ex.http_status : nil,
          duration_ms: started && elapsed_ms(started),
        )
        raise # dispatch_to_channel still classifies this as :error (unchanged)
      end

      # objid first (custid == objid for every modern row), then public extid.
      # Unresolvable → nil: no tenant to kill, and the log row carries user_id nil.
      def resolve_webhook_customer(custid)
        return nil if custid.to_s.empty?
        cust = Onetime::Customer.load(custid.to_s)
        cust = Onetime::Customer.find_by_extid(custid.to_s) unless cust&.exists?
        cust&.exists? ? cust : nil
      rescue StandardError => ex
        logger.warn 'Webhook customer lookup failed', error: ex.class.name
        nil
      end

      def elapsed_ms(started)
        ((Process.clock_gettime(Process::CLOCK_MONOTONIC) - started) * 1000).round
      end
```

```ruby
require 'onetime/models/colonel_audit_event'

module Auth::Operations::Customers
  class SetWebhooksDisabled
    include Onetime::LoggerMethods
    include Onetime::AuditedFailure
    include Onetime::AuditReason

    AUDIT_VERB_DISABLE = 'customer.webhooks.disable'
    AUDIT_VERB_ENABLE  = 'customer.webhooks.enable'

    audit_failures :call,
      verb: -> { @disabled ? AUDIT_VERB_DISABLE : AUDIT_VERB_ENABLE },
      target: -> { @customer&.extid }

    Result = Data.define(:status, :customer, :disabled)   # status :success | :no_change

    def initialize(customer:, disabled:, actor:, reason: nil)
      @customer = customer
      @disabled = disabled ? true : false
      @actor    = actor
      @reason   = normalize_reason(reason)
    end

    def call
      if @customer.webhooks_disabled? == @disabled
        record_no_change_event
        return Result.new(status: :no_change, customer: @customer, disabled: @disabled)
      end

      @customer.webhooks_disabled = @disabled     # native boolean; never ||=
      @customer.save

      # Fail-OPEN (not fail_closed): the flag is inspectable on the customer row
      # and in GET /users/:id, so an unrecorded flip is reconstructible.
      Onetime::ColonelAuditEvent.record(
        actor: @actor, verb: verb, target: @customer.extid, result: :success,
        detail: with_reason(disabled: @disabled),
      )
      auth_logger.debug "[#{verb}] #{@customer.extid}"
      Result.new(status: :success, customer: @customer, disabled: @disabled)
    end

    private

    def verb = @disabled ? AUDIT_VERB_DISABLE : AUDIT_VERB_ENABLE

    # #4337: the attempt is the fact; same verb, outcome no_change, reason rides.
    def record_no_change_event
      Onetime::ColonelAuditEvent.record(
        actor: @actor, verb: verb, target: @customer.extid, result: :success,
        detail: with_reason(outcome: 'no_change', disabled: @disabled),
      )
    end
  end
end
```

# DECISIONS
1. **Issue premise is wrong about the code (confirmed):** there is no webhook-target model, field, or list to show. `webhook_url` exists only inside the queue payload (`dispatch_notification.rb:194`, `notification_worker.rb:22` comment) and nothing in production publishes it (only `bin/ots queue ping` publishes to `notifications.alert.push`, without a URL). "List configured targets, where they point" therefore cannot be built; the delivery log IS the target inventory (host per delivery, per customer). Scope file's correction stands.

2. **Storage pattern: `EmailSuppression`-style `class_sorted_set` + trim, not the raw `lpush/ltrim` ring at `dispatch_notification.rb:132–136`.** The sorted set gives offset pagination (`revrange`) and `element_count` for the frozen envelope with the same cap semantics, and it is the pattern the existing paginated colonel list (`list_email_deliverability_events.rb`) already reads. MAX 500 as scoped.

3. **Missing `webhook_url` is NOT logged.** Nothing is attempted and there is no host; `bin/ots queue ping` would otherwise spam the ring. Only real attempts and kill-switch skips are recorded (`status: skipped, reason: 'disabled'`).

4. **Customer resolution from the payload:** `Onetime::Customer.load(custid)` (custid == objid, `customer.rb:222`) then `find_by_extid`. Unresolvable → deliver anyway, log `user_id: nil`, no stats. The log stores the PUBLIC extid, never an objid (audit-trail rule applied to this store too).

5. **Kill-switch check runs after the URL-presence check and before any DNS/HTTP work** — a disabled tenant costs zero egress.

6. **Non-2xx path:** replaced the bare `RuntimeError` at :204–206 with `WebhookResponseError` carrying `http_status`, message text unchanged, so the log has the upstream code. `dispatch_to_channel`'s `:error` classification is untouched; the existing integration spec still passes.

7. **Field encoding:** `boolean_field :webhooks_disabled, storage: :native` (the type's own docs say `:native` for new fields; `nil` = pre-feature row = enabled). Read only via `webhooks_disabled?` — the raw-read guard spec would otherwise flag a bare read. No `webhooks_disabled_at/by/reason` stamps (scope says boolean only); the audit trail is the who/when/why record, filterable with verb prefix `customer.webhooks` on the existing audit screen.
